cmake_minimum_required(VERSION 3.16)
project(sdn_simulator LANGUAGES CXX)

add_library(network src/Network.cpp src/Router.cpp)
target_include_directories(network PUBLIC src)
target_compile_features(network PUBLIC cxx_std_17)
set_target_properties(network PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(network PRIVATE /W4)
else()
    target_compile_options(network PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(sdn-simulator src/main.cpp)
target_link_libraries(sdn-simulator PRIVATE network)
configure_file(src/network.txt "${CMAKE_CURRENT_BINARY_DIR}/network.txt" COPYONLY)

include(CTest)
if(BUILD_TESTING)
    add_executable(network-tests tests/network_tests.cpp)
    target_link_libraries(network-tests PRIVATE network)
    add_test(NAME network-tests COMMAND network-tests "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
